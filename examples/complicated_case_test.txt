//  - 뒤늦게 추가된 기능 위주: 정적 배열 / 함수 / 상수 폴딩 최적화
//  - 코너케이스 포함. 런타임 에러 없이 끝까지 실행되도록 구성.
//  각 print 옆 주석은 기대 출력값.
// ============================================================


// ── 1. 상수 연산 최적화 (Checker 단계에서 폴딩됨) ──────────────
// 아래 식들은 실행 전에 미리 하나의 상수로 접힘.
print 1 * 2 * 3 * 4 * 5 / 6 + 7 + 8 + 9;   // 44
print (2 + 3) * (10 - 4);                  // 30  (그룹핑 폴딩)
print "Code" + "Fab" + " v" + "1.0";       // CodeFab v1.0  (문자열 연결 폴딩)
print 10 > 3 and 5 <= 5;                   // true  (비교+논리 폴딩)
print !false;                              // true  (단항 폴딩)
print -(-42);                              // 42    (중첩 단항 폴딩)
print 7 / 2;                               // 3.5   (비정수 double은 .0 없이 그대로)


// ── 2. 연산자 우선순위 코너케이스 ───────────────────────────────
// and 가 or 보다 강하게 결합: ((1<2) and (3>4)) or (5==5)
print 1 < 2 and 3 > 4 or 5 == 5;           // true
print !(1 == 1);                           // false  (그룹 전체 부정)


// ── 3. 정적 배열: 선언/기본값/쓰기/읽기 ─────────────────────────
// 배열 원소는 nil(null)로 초기화됨.
var slots[3];
print slots;                               // [nil, nil, nil]  (초기 기본값 코너케이스)

slots[0] = 10;
slots[2] = 30;                             // 가운데(1번)는 비워둠
print slots;                               // [10, nil, 30]

// 크기를 식으로 지정 (2+3 → 5), 그리고 Array() 형태도 동일하게 동작
var grid = Array(2 + 3);
print grid;                                // [nil, nil, nil, nil, nil]


// ── 4. 배열 + 반복문: 채우고, 계산식 인덱스로 접근, 합산 ──────────
var data[5];
for (var k = 0; k < 5; k = k + 1) {
    data[k] = k * k;                       // 제곱으로 채움
}
print data;                                // [0, 1, 4, 9, 16]

data[2 + 1] = 99;                          // 계산식 인덱스로 쓰기
print data[3];                             // 99

var total = 0;
for (var k = 0; k < 5; k = k + 1) {
    total = total + data[k];
}
print total;                               // 120  (0+1+4+99+16)


// ── 5. 함수: 매개변수 / 반환 / 재귀 ────────────────────────────
Func square(x) {
    return x * x;
}
print square(9);                           // 81

// 재귀 - 팩토리얼 (본문 최상위에 return 존재)
Func fact(n) {
    if (n <= 1) {
        return 1;
    }
    return n * fact(n - 1);
}
print fact(5);                             // 120

// 재귀 - 피보나치
Func fib(n) {
    if (n < 2) {
        return n;
    }
    return fib(n - 1) + fib(n - 2);
}
print fib(10);                             // 55

// 함수 결과를 배열에 저장
var fibs[6];
for (var i = 0; i < 6; i = i + 1) {
    fibs[i] = fib(i);
}
print fibs;                                // [0, 1, 1, 2, 3, 5]


// ── 6. 스코프 코너케이스: 반복문 변수 섀도잉 ────────────────────
var i = 100;                               // 바깥 i
for (var i = 0; i < 2; i = i + 1) {        // 루프 스코프의 i (섀도잉)
    for (var j = 0; j < 2; j = j + 1) {
        print i * 10 + j;                  // 0, 1, 10, 11
    }
}
print i;                                   // 100  (바깥 i는 영향 없음)


// ── 7. 댕글링 else 코너케이스 ──────────────────────────────────
// else 는 가장 가까운 (안쪽) if 에 결합됨.
if (true) if (false) print "inner-if"; else print "inner-else";  // inner-else


// ── 8. 마무리 ─────────────────────────────────────────────────
var name = "CodeFab";
if (square(2) + fact(3) == 10 and fib(7) == 13) {
    print name + " 통합 예제 통과!";        // CodeFab 통합 예제 통과!
} else {
    print name + " 검증 실패";
}
